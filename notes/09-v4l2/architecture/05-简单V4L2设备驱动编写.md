# 简单 V4L2 设备驱动编写

本文面向“不需要复杂 media graph”的 capture 设备：

- 虚拟摄像头。
- 一个硬件块直接 DMA 到内存。
- 被一个驱动整体封装的 USB/PCI 设备。

代码以 Linux 4.9 风格表达。厂商 4.9 树常有 API 差异，实际编译必须对照目标内核头文件。

## 1. 驱动边界

```text
V4L2 Core 提供：
  字符设备入口、ioctl 分发、标准 ABI

videobuf2 提供：
  buffer 分配、映射、状态、QBUF/DQBUF/poll

本驱动提供：
  支持哪些格式
  如何配置硬件
  如何把 buffer 交给硬件
  如何启动/停止
  一帧何时完成
```

## 2. 私有对象

```c
struct my_buffer {
    struct vb2_v4l2_buffer vb;
    struct list_head list;
};

struct my_video {
    struct device *dev;

    struct v4l2_device v4l2_dev;
    struct video_device vdev;
    struct v4l2_ctrl_handler ctrl_hdl;
    struct vb2_queue vb_queue;

    struct mutex lock;       /* ioctl/queue 串行化 */
    spinlock_t qlock;        /* 中断与进程共享 active list */
    struct list_head active;

    u32 width;
    u32 height;
    u32 pixelformat;
    u32 bytesperline;
    u32 sizeimage;

    unsigned int sequence;
    bool streaming;

    /* 时钟、寄存器、IRQ、DMA channel、USB 对象等 */
};
```

对象关系应保持清晰：

```text
my_video
 ├─ v4l2_device：实例容器
 ├─ video_device：/dev/videoX
 ├─ vb2_queue：该节点的 capture queue
 ├─ ctrl_handler：参数集合
 └─ active：已交给驱动、等待硬件处理的 buffer
```

## 3. 格式状态

先只支持一种格式，减少无关复杂度：

```c
static void my_fill_pix_format(struct my_video *dev,
                               struct v4l2_pix_format *pix)
{
    pix->width        = dev->width;
    pix->height       = dev->height;
    pix->pixelformat  = dev->pixelformat;
    pix->field        = V4L2_FIELD_NONE;
    pix->bytesperline = dev->bytesperline;
    pix->sizeimage    = dev->sizeimage;
    pix->colorspace   = V4L2_COLORSPACE_SRGB;
}

static void my_clamp_format(struct v4l2_pix_format *pix)
{
    pix->width  = clamp_t(u32, pix->width, 160, 1920);
    pix->height = clamp_t(u32, pix->height, 120, 1080);
    pix->width  &= ~1; /* 例如 YUYV 要求偶数宽度 */

    pix->pixelformat  = V4L2_PIX_FMT_YUYV;
    pix->field        = V4L2_FIELD_NONE;
    pix->bytesperline = pix->width * 2;
    pix->sizeimage    = pix->bytesperline * pix->height;
    pix->colorspace   = V4L2_COLORSPACE_SRGB;
}
```

ioctl 回调：

```c
static int my_querycap(struct file *file, void *priv,
                       struct v4l2_capability *cap)
{
    strlcpy(cap->driver, "my-video", sizeof(cap->driver));
    strlcpy(cap->card, "My Capture Device", sizeof(cap->card));
    strlcpy(cap->bus_info, "platform:my-video",
            sizeof(cap->bus_info));
    cap->device_caps = V4L2_CAP_VIDEO_CAPTURE |
                       V4L2_CAP_STREAMING;
    cap->capabilities = cap->device_caps |
                        V4L2_CAP_DEVICE_CAPS;
    return 0;
}

static int my_enum_fmt(struct file *file, void *priv,
                       struct v4l2_fmtdesc *f)
{
    if (f->index != 0)
        return -EINVAL;
    f->pixelformat = V4L2_PIX_FMT_YUYV;
    strlcpy(f->description, "YUYV 4:2:2",
            sizeof(f->description));
    return 0;
}

static int my_g_fmt(struct file *file, void *priv,
                    struct v4l2_format *f)
{
    struct my_video *dev = video_drvdata(file);

    my_fill_pix_format(dev, &f->fmt.pix);
    return 0;
}

static int my_try_fmt(struct file *file, void *priv,
                      struct v4l2_format *f)
{
    my_clamp_format(&f->fmt.pix);
    return 0;
}

static int my_s_fmt(struct file *file, void *priv,
                    struct v4l2_format *f)
{
    struct my_video *dev = video_drvdata(file);
    int ret;

    if (vb2_is_busy(&dev->vb_queue))
        return -EBUSY;

    ret = my_try_fmt(file, priv, f);
    if (ret)
        return ret;

    dev->width        = f->fmt.pix.width;
    dev->height       = f->fmt.pix.height;
    dev->pixelformat  = f->fmt.pix.pixelformat;
    dev->bytesperline = f->fmt.pix.bytesperline;
    dev->sizeimage    = f->fmt.pix.sizeimage;
    return 0;
}
```

## 4. vb2 ops

### 4.1 `queue_setup`

```c
static int my_queue_setup(struct vb2_queue *vq,
                          const void *parg,
                          unsigned int *nbufs,
                          unsigned int *nplanes,
                          unsigned int sizes[],
                          void *alloc_ctxs[])
{
    struct my_video *dev = vb2_get_drv_priv(vq);

    if (*nplanes) {
        if (sizes[0] < dev->sizeimage)
            return -EINVAL;
        return 0;
    }

    *nplanes = 1;
    sizes[0] = dev->sizeimage;
    if (*nbufs < 3)
        *nbufs = 3;
    return 0;
}
```

### 4.2 `buf_prepare`

```c
static int my_buf_prepare(struct vb2_buffer *vb)
{
    struct my_video *dev = vb2_get_drv_priv(vb->vb2_queue);

    if (vb2_plane_size(vb, 0) < dev->sizeimage)
        return -EINVAL;

    vb2_set_plane_payload(vb, 0, dev->sizeimage);
    return 0;
}
```

对压缩格式，最终 `bytesused` 可能在帧完成时才能确定，prepare 阶段不应伪造固定 payload。

### 4.3 `buf_queue`

```c
static void my_buf_queue(struct vb2_buffer *vb)
{
    struct my_video *dev = vb2_get_drv_priv(vb->vb2_queue);
    struct my_buffer *buf =
        container_of(to_vb2_v4l2_buffer(vb),
                     struct my_buffer, vb);
    unsigned long flags;

    spin_lock_irqsave(&dev->qlock, flags);
    list_add_tail(&buf->list, &dev->active);
    spin_unlock_irqrestore(&dev->qlock, flags);

    my_kick_dma_if_idle(dev);
}
```

### 4.4 `start_streaming`

```c
static int my_start_streaming(struct vb2_queue *vq,
                              unsigned int count)
{
    struct my_video *dev = vb2_get_drv_priv(vq);
    int ret;

    dev->sequence = 0;

    ret = my_hw_start(dev);
    if (ret) {
        my_return_all_buffers(dev, VB2_BUF_STATE_QUEUED);
        return ret;
    }

    dev->streaming = true;
    my_kick_dma_if_idle(dev);
    return 0;
}
```

启动失败时必须归还已经通过 `buf_queue` 交给驱动的 buffer。

### 4.5 `stop_streaming`

```c
static void my_stop_streaming(struct vb2_queue *vq)
{
    struct my_video *dev = vb2_get_drv_priv(vq);

    dev->streaming = false;
    my_hw_stop(dev);
    synchronize_irq(my_irq(dev));
    my_return_all_buffers(dev, VB2_BUF_STATE_ERROR);
}
```

`synchronize_irq()` 是否适用取决于实际硬件。目标是保证 stop 返回后不会再有完成路径访问旧 buffer。

### 4.6 ops 表

```c
static const struct vb2_ops my_vb2_ops = {
    .queue_setup     = my_queue_setup,
    .buf_prepare     = my_buf_prepare,
    .buf_queue       = my_buf_queue,
    .start_streaming = my_start_streaming,
    .stop_streaming  = my_stop_streaming,
    .wait_prepare    = vb2_ops_wait_prepare,
    .wait_finish     = vb2_ops_wait_finish,
};
```

## 5. 完成一帧

硬件中断伪代码：

```c
static irqreturn_t my_irq_handler(int irq, void *data)
{
    struct my_video *dev = data;
    struct my_buffer *buf;
    unsigned long flags;

    if (!my_hw_frame_done(dev))
        return IRQ_NONE;

    spin_lock_irqsave(&dev->qlock, flags);
    if (list_empty(&dev->active)) {
        spin_unlock_irqrestore(&dev->qlock, flags);
        my_hw_ack_irq(dev);
        return IRQ_HANDLED;
    }

    buf = list_first_entry(&dev->active,
                           struct my_buffer, list);
    list_del(&buf->list);
    spin_unlock_irqrestore(&dev->qlock, flags);

    my_hw_ack_irq(dev);

    vb2_set_plane_payload(&buf->vb.vb2_buf, 0,
                          dev->sizeimage);
    buf->vb.sequence = dev->sequence++;
    buf->vb.field = V4L2_FIELD_NONE;
    buf->vb.vb2_buf.timestamp = ktime_get_ns();

    vb2_buffer_done(&buf->vb.vb2_buf,
                    VB2_BUF_STATE_DONE);

    my_kick_dma_if_idle(dev);
    return IRQ_HANDLED;
}
```

真实 DMA 驱动不能只在完成中断后才“随便取链表第一个”；应明确记录当前 DMA buffer，并在安全点编程下一块地址。

虚拟摄像头可在定时器或 kthread 中取得 active buffer，使用 `vb2_plane_vaddr()` 填图，再调用 `vb2_buffer_done()`。

## 6. 归还全部 buffer

安全写法是先摘到临时链表，再在锁外完成：

```c
static void my_return_all_buffers(struct my_video *dev,
                                  enum vb2_buffer_state state)
{
    LIST_HEAD(tmp);
    struct my_buffer *buf, *next;
    unsigned long flags;

    spin_lock_irqsave(&dev->qlock, flags);
    list_splice_init(&dev->active, &tmp);
    spin_unlock_irqrestore(&dev->qlock, flags);

    list_for_each_entry_safe(buf, next, &tmp, list) {
        list_del(&buf->list);
        vb2_buffer_done(&buf->vb.vb2_buf, state);
    }
}
```

    .vidioc_dqbuf             = vb2_ioctl_dqbuf,

    .vidioc_streamoff         = vb2_ioctl_streamoff,

static const struct v4l2_file_operations my_fops = {
    .owner          = THIS_MODULE,
    .open           = v4l2_fh_open,
    .release        = vb2_fop_release,
    .read           = vb2_fop_read,

如果 queue 只声明 `VB2_MMAP`，就不应广告 read 能力，也可以不提供 `.read`。

## 8. 初始化 vb2 queue

```c
static int my_init_queue(struct my_video *dev)
{
    struct vb2_queue *q = &dev->vb_queue;

    memset(q, 0, sizeof(*q));
    q->type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    q->io_modes = VB2_MMAP | VB2_DMABUF;
    q->drv_priv = dev;
    q->buf_struct_size = sizeof(struct my_buffer);
    q->ops = &my_vb2_ops;
    q->mem_ops = &vb2_dma_contig_memops; /* 按硬件选择 */
    q->timestamp_flags = V4L2_BUF_FLAG_TIMESTAMP_MONOTONIC;
    q->lock = &dev->lock;
    q->dev = dev->dev;

    return vb2_queue_init(q);
}
```

虚拟设备使用 `vb2_vmalloc_memops` 更简单；DMA 设备应根据硬件和 IOMMU 选择 dma-contig 或 dma-sg。

## 9. 初始化 video device

```c
static const struct video_device my_vdev_template = {
    .name       = "my-video",
    .fops       = &my_fops,
    .ioctl_ops  = &my_ioctl_ops,
    .release    = video_device_release_empty,
    .device_caps = V4L2_CAP_VIDEO_CAPTURE |
                   V4L2_CAP_STREAMING,
};

static int my_register_video(struct my_video *dev)
{
    int ret;

    ret = v4l2_device_register(dev->dev, &dev->v4l2_dev);
    if (ret)
        return ret;

    ret = my_init_controls(dev);
    if (ret)
        goto err_v4l2;

    ret = my_init_queue(dev);
    if (ret)
        goto err_ctrl;

    dev->vdev = my_vdev_template;
    dev->vdev.v4l2_dev = &dev->v4l2_dev;
    dev->vdev.queue = &dev->vb_queue;
    dev->vdev.lock = &dev->lock;
    dev->vdev.ctrl_handler = &dev->ctrl_hdl;

    video_set_drvdata(&dev->vdev, dev);

    ret = video_register_device(&dev->vdev,
                                VFL_TYPE_GRABBER, -1);
    if (ret)
        goto err_ctrl;

    return 0;

err_ctrl:
    v4l2_ctrl_handler_free(&dev->ctrl_hdl);
err_v4l2:
    v4l2_device_unregister(&dev->v4l2_dev);
    return ret;
}
```

## 10. probe 的完整顺序

```text
匹配 platform/USB/PCI 设备
→ devm_kzalloc 私有对象
→ 初始化 mutex、spinlock、active list
→ 获取 clock/regulator/GPIO/MMIO/IRQ/DMA
→ 设置默认格式
→ v4l2_device_register
→ controls init
→ vb2_queue_init
→ 填写并注册 video_device
→ 允许 runtime PM / 返回成功
```

remove：

```text
阻止新操作
→ video_unregister_device
→ 停硬件/释放 IRQ、DMA 等资源
→ controls free
→ v4l2_device_unregister
```

devm 资源和非 devm 资源的生命周期必须分清。

## 11. 最常见的驱动错误

1. `S_FMT` 后没保存驱动修正后的格式。
2. `queue_setup` 的 size 与 `sizeimage` 不一致。
3. QBUF 后仍让 APP/CPU 改数据。
4. 中断完成时没设置 payload。
5. streamoff 没归还全部 active buffer。
6. 启动失败后遗留 ACTIVE buffer。
7. active list 在进程和 IRQ 间没有自旋锁。
8. 在持 spinlock 时做睡眠操作。
9. remove 时 video node 仍可能进入硬件。
10. 把 `video_device` 与 `v4l2_device` 当成同一个对象。
